#!/usr/bin/env python3
import subprocess
import sqlite3
import uuid
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel

router = APIRouter()
DB_PATH = "/home/james/SovereignOS/dna/sovereign_now.db"

class VoiceHealPayload(BaseModel):
    transcript: str
    pilot_node: str

@router.post("/api/system/heal/voice")
def execute_voice_heal(payload: VoiceHealPayload):
    text_lower = payload.transcript.lower()
    target_service = None
    service_cmd = None
    
    if "bistro" in text_lower or "port 3015" in text_lower:
        target_service = "James's Bistro Portal"
        service_cmd = "sudo systemctl restart sovereign-bistro.service"
    elif "chat" in text_lower or "personas" in text_lower:
        target_service = "M.A.R.D Chatbot Engine"
        service_cmd = "sudo systemctl restart sovereign-chatbots.service"
    elif "sniper" in text_lower or "stream" in text_lower:
        target_service = "Sovereign Stream Relay"
        service_cmd = "sudo systemctl restart sovereign-stream-relay.service"
        
    if not target_service:
        raise HTTPException(status_code=422, detail="No matching service anomaly detected in transcript.")
        
    inc_number = f"INC{uuid.uuid4().hex[:8].upper()}"
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()
    
    cursor.execute("""
        INSERT INTO sovereign_tickets (number, short_description, state, type, sys_created_on, sys_updated_on)
        VALUES (?, ?, 1, 'INC', datetime('now'), datetime('now'))
    """, (inc_number, f"Voice-Triggered Healing Sequence for {target_service}",))
    conn.commit()
    conn.close()
    
    try:
        subprocess.run(service_cmd, shell=True, check=True, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        return {
            "status": "HEALED",
            "incident_logged": inc_number,
            "resolved_service": target_service,
            "execution_trail": f"Redirected process output to logs successfully."
        }
    except subprocess.CalledProcessError:
        raise HTTPException(status_code=500, detail=f"Bare-metal hardware command failed during execution loop.")
