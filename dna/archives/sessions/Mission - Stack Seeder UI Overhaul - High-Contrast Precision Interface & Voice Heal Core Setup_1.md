import os
import sqlite3
from fastapi import APIRouter, HTTPException

router = APIRouter()
DB_PATH = "/home/james/SovereignOS/dna/sovereign_now.db"

@router.get("/api/system/seeder/assets")
def get_seeder_assets():
    # Connect directly to the production configuration management database
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()
    
    try:
        # Query the media tracking table to fetch all active brand assets
        cursor.execute("SELECT sys_id, brand_id, asset_tag, file_name, content_type FROM sys_media_asset ORDER BY sys_created_on DESC")
        assets = cursor.fetchall()
        
        asset_list = []
        for row in assets:
            asset_list.append({
                "sys_id": row[0],
                "brand_id": row[1],
                "asset_tag": row[2],
                "file_name": row[3],
                "content_type": row[4],
                "url": f"/api/now/media/file/{row[0]}"
            })
            
        return {"status": "SUCCESS", "count": len(asset_list), "assets": asset_list}
    except sqlite3.OperationalError:
        raise HTTPException(status_code=500, detail="Relational database table structure missing or locked.")
    finally:
        conn.close()
